import enum
from datetime import datetime, timezone
from typing import Annotated

from sqlmodel import Field, Relationship, Session, SQLModel, create_engine

sqlite_file_name = "database.db"
connect_args = {"check_same_thread": False}
engine = create_engine(f"sqlite:///{sqlite_file_name}", connect_args=connect_args)


def create_db_and_tables():
    SQLModel.metadata.create_all(engine)


def get_session():
    with Session(engine) as session:
        yield session


class UserBase(SQLModel):
    username: Annotated[str, Field(index=True, unique=True)]
    email: Annotated[str, Field(index=True, unique=True)]


class User(UserBase, table=True):
    id: Annotated[int | None, Field(primary_key=True)] = None
    documents: list["Document"] = Relationship(back_populates="user")
    password: str


class UserPublic(UserBase):
    id: int


class UserCreate(UserBase):
    password: str


class DocumentStatus(str, enum.Enum):
    PROCESSING = "processing"
    STORED = "stored"
    FAILED = "failed"


class Document(SQLModel, table=True):
    id: Annotated[int | None, Field(primary_key=True)] = None
    filename: Annotated[str, Field(index=True)]
    user_id: Annotated[int, Field(foreign_key="user.id")]
    user: User = Relationship(back_populates="documents")
    vector_namespace: str
    uploaded_at: Annotated[
        datetime, Field(default_factory=lambda: datetime.now(timezone.utc))
    ]
    upload_status: DocumentStatus = DocumentStatus.PROCESSING


if __name__ == "__main__":
    create_db_and_tables()

import logging
from datetime import datetime, timedelta, timezone
from typing import Annotated

import jwt
from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer, OAuth2PasswordRequestForm
from pwdlib import PasswordHash
from sqlmodel import Session, select

from app.database import User, UserCreate, UserPublic, get_session
from app.settings import Token, get_sys_settings

router = APIRouter()


password_hash = PasswordHash.recommended()


@router.post(
    "/register", response_model=UserPublic, status_code=status.HTTP_201_CREATED
)
def register_user(user: UserCreate, session: Annotated[Session, Depends(get_session)]):
    statement = select(User).where(
        (User.username == user.username) | (User.email == user.email)
    )
    existing_user = session.exec(statement).first()
    if existing_user:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="A user with this username or email already exists.",
        )
    user.password = password_hash.hash(user.password)
    db_user = User.model_validate(user)
    session.add(db_user)
    session.commit()
    return db_user


def authenticate_user(
    session: Session,
    username_or_email: str,
    password: str,
):
    statement = select(User).where(
        (User.username == username_or_email) | (User.email == username_or_email)
    )
    user = session.exec(statement).first()
    if user:
        if password_hash.verify(password, user.password):
            return user
    else:
        password_hash.verify(password, get_sys_settings().dummy_pwd)
    raise HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Incorrect username, email, or password.",
        headers={"WWW-Authenticate": "bearer"},
    )


def create_token(data_dict: dict, expires_delta: timedelta = timedelta(minutes=30)):
    data_dict = data_dict.copy()
    expires = datetime.now(timezone.utc) + expires_delta
    data_dict.update({"exp": expires})
    encoded_jwt = jwt.encode(
        data_dict,
        get_sys_settings().token_secret_key,
        algorithm=get_sys_settings().token_encryption_algorithm,
    )
    return encoded_jwt


@router.post("/token")
def users_login(
    session: Annotated[Session, Depends(get_session)],
    form_data: Annotated[OAuth2PasswordRequestForm, Depends()],
):
    user = authenticate_user(session, form_data.username, form_data.password)
    access_token = create_token({"sub": user.username})
    return Token(access_token=access_token, token_type="bearer")


# Extract token from HTTP header
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/token")


async def get_cur_user(
    token: Annotated[str, Depends(oauth2_scheme)],
    session: Annotated[Session, Depends(get_session)],
):
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(
            token,
            get_sys_settings().token_secret_key,
            algorithms=get_sys_settings().token_encryption_algorithm,
        )
        username = payload["sub"]
        statement = select(User).where(User.username == username)
        user = session.exec(statement).first()
        return user

    except Exception as e:
        logging.error(e)
        raise credentials_exception


@router.get("/test", response_model=UserPublic)
def get_info_test(user: Annotated[User, Depends(get_cur_user)]):
    return user

# 📝 Text/Markdown (.txt, .md): TextLoader or UnstructuredMarkdownLoader
# 🌐 HTML (.html): BSHTMLLoader
# 📕 PDF (.pdf): PyPDFLoader
# 📘 Word (.doc, .docx): UnstructuredWordDocumentLoader
# 🖼️ Images (.jpg, .png): UnstructuredImageLoader (which handles basic OCR)
import os
from typing import Iterator
from langchain_google_genai import GoogleGenerativeAIEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_core.documents import Document
from langchain_community.document_loaders import (
    UnstructuredMarkdownLoader,
    BSHTMLLoader,
    PyPDFLoader,
    UnstructuredWordDocumentLoader,
    UnstructuredImageLoader,
)
from langchain_community.vectorstores import Chroma


def file_embedding(files: list[str], vector_database: str) -> Chroma:
    generators: list[Iterator[Document]] = []
    for file in files:
        match os.path.splitext(file)[-1]:
            case ".pdf":
                loader = PyPDFLoader(file)
            case ".md" | ".txt":
                loader = UnstructuredMarkdownLoader(file)
            case ".html":
                loader = BSHTMLLoader(file)
            case ".doc" | ".docx":
                loader = UnstructuredWordDocumentLoader(file)
            case ".png" | ".jpg":
                loader = UnstructuredImageLoader(file)
        generators.append(loader.lazy_load())
    text_splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    embeddings_model = GoogleGenerativeAIEmbeddings(model="models/text-embedding-004")
    docs: list[Document] = []
    for generator in generators:
        docs.extend(text_splitter.split_documents(generator))
    db = Chroma.from_documents(
        documents=docs,
        embedding=embeddings_model,
        persist_directory=vector_database,
    )
    # we also need to save data into local files and many other pending opeartions here.
    return db

import os
from aiofiles.tempfile import NamedTemporaryFile
from fastapi import Depends, APIRouter, BackgroundTasks, UploadFile
from app.api.convert import file_embedding
from app.dependencies import SysSettings, get_sys_settings

router = APIRouter()


# This is cpu bounded, so I use a blocking function here. This allows a separate worker thread for BackgroundTasks.
def process_data(files, vector_database):
    try:
        file_embedding(files, vector_database)
    finally:
        for file in files:
            os.remove(file)


# I need a well rounded api routing design.
@router.post("/upload")
async def file_upload(
    files: list[UploadFile],
    background_task: BackgroundTasks,
    sys_path: SysSettings = Depends(get_sys_settings),
):

    temp_files = []
    for file in files:
        async with NamedTemporaryFile(
            "wb", dir=sys_path.temp_file, delete=False
        ) as buffer:
            while chunk := await file.read(1024 * 1024):
                await buffer.write(chunk)
            await buffer.flush()
            temp_files.append(buffer.name)
        background_task.add_task(process_data, temp_files, sys_path.vector_database)

    return {"message": "Upload started"}
