22:def vectors_embedding(file_path: str, document_id: str, user_id: str):
23:    file_type = os.path.splitext(file_path)[-1].lower()
24:
25:    # 1. Route to the correct Loader AND Splitter
26:    match file_type:
27:        case ".md":
28:            loader = TextLoader(file_path)
29:            headers_to_split_on = [
30:                ("#", "Header 1"),
31:                ("##", "Header 2"),
32:                ("###", "Header 3"),
33:            ]
34:            raw_text = loader.load()[0].page_content
35:            docs = MarkdownHeaderTextSplitter(
36:                headers_to_split_on=headers_to_split_on
37:            ).split_text(raw_text)
38:
39:        case ".html":
40:            headers_to_split_on = [("h1", "Header 1"), ("h2", "Header 2")]
41:            splitter = HTMLHeaderTextSplitter(headers_to_split_on=headers_to_split_on)
42:            docs = splitter.split_text_from_file(file_path)
43:
44:        case ".pdf" | ".txt" | ".rst" | ".doc" | ".docx" | ".png" | ".jpg":
45:            if file_type == ".pdf":
46:                loader = PyPDFLoader(file_path)
47:            elif file_type in (".txt", ".rst"):
48:                loader = TextLoader(file_path)
49:            elif file_type in [".doc", ".docx"]:
50:                loader = UnstructuredWordDocumentLoader(file_path)
51:            else:
52:                loader = UnstructuredImageLoader(file_path)
53:
54:            docs = loader.load()
55:
56:        case _:
57:            raise ValueError(f"Unsupported file type: {file_type}")
58:
59:    splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=150)
60:    chunks = splitter.split_documents(docs)
61:
62:    # 2. Inject Metadata
63:    for chunk in chunks:
64:        # Markdown splitters create 'metadata', standard splitters retain existing metadata
65:        if "source" not in chunk.metadata:
66:            chunk.metadata["source"] = file_path
67:        chunk.metadata["document_id"] = str(document_id)
68:
69:    # 3. Embed and Store
70:    embedding_service = settings.services.embedding_model
71:
72:    embeddings_model = get_embeddings(**embedding_service.model_dump())
73:
74:    Chroma.from_documents(
75:        client=engine.chroma_client,
76:        documents=chunks,
77:        embedding=embeddings_model,
78:        collection_name=f"user-{user_id}",
79:    )

14:def compute_metrics(results: dict[str, list[dict]]) -> dict[str, dict]:
15:    """Compute all metrics for each variant."""
16:    report = {}
17:
18:    for variant, items in results.items():
19:        total = len(items)
20:        recall_scores = []
21:        mrr_scores = []
22:        citation_scores = []
23:        refusal_correct = 0
24:        refusal_total = 0
25:        latencies = []
26:        model_counts = {}
27:
28:        for item in items:
29:            # Retrieval metrics
30:            recall_scores.append(
31:                recall_at_k(item["sources"], item["expected_sources"], k=RETRIEVAL_K)
32:            )
33:            mrr_scores.append(mrr(item["sources"], item["expected_sources"]))
34:
35:            # Citation accuracy
36:            citation_scores.append(
37:                citation_accuracy(
38:                    item["answer"], item["sources"], item["expected_sources"]
39:                )
40:            )
41:
42:            # Refusal accuracy
43:            if item["out_of_scope"]:
44:                refusal_total += 1
45:                if refusal_accuracy(item["answer"], item["out_of_scope"]):
46:                    refusal_correct += 1
47:            else:
48:                # In-scope should NOT refuse
49:                if refusal_accuracy(item["answer"], item["out_of_scope"]):
50:                    refusal_correct += 1
51:                refusal_total += 1
52:
53:            # Latency
54:            latencies.append(item["latency_ms"])
55:
56:            # Model usage
57:            model = item.get("metadata", {}).get("selected_model", "default")
58:            model_counts[model] = model_counts.get(model, 0) + 1
59:
60:        report[variant] = {
61:            "Recall@4": round(sum(recall_scores) / total, 3) if total else 0,
62:            "MRR": round(sum(mrr_scores) / total, 3) if total else 0,
63:            "Citation Acc.": round(sum(citation_scores) / total, 3) if total else 0,
64:            "Refusal Acc.": round(refusal_correct / refusal_total, 3) if refusal_total else 0,
65:            "Avg Latency (ms)": round(sum(latencies) / total, 0) if total else 0,
66:            "Model Distribution": model_counts,
67:        }

69:    return report

36:    query: str,
37:    variant: str,
38:) -> dict:
39:    """Execute a single RAG query with a specific variant configuration.
40:
41:    Returns dict with answer, sources, latency_ms, and metadata.
42:    """
43:    use_hyde = variant in (VARIANT_HYDE, VARIANT_HYDE_ROUTING)
44:    use_routing = variant in (VARIANT_ROUTING, VARIANT_HYDE_ROUTING)
45:
46:    start = time.perf_counter()
47:    metadata = {"variant": variant, "hyde": use_hyde, "routing": use_routing}
48:
49:    # 1. Setup vectorstore
50:    embedding_config = settings.services.embedding_model.model_dump()
51:    embeddings_model = get_embeddings(**embedding_config)
52:    vectorstore = Chroma(
53:        client=engine.chroma_client,
54:        collection_name=EVAL_COLLECTION,
55:        embedding_function=embeddings_model,
56:    )
57:
58:    # 2. HyDE
59:    search_text = query
60:    if use_hyde:
61:        router_config = settings.services.router_model.model_dump()
62:        router_llm = get_chat_model(**router_config, temperature=settings.hyde.temperature)
63:        search_text = await generate_hypothetical_document(query, router_llm)
64:        metadata["hyde_document"] = search_text[:200]
65:
66:    # 3. Retrieve
67:    if use_routing:
68:        results = vectorstore.similarity_search_with_relevance_scores(
69:            search_text, k=RETRIEVAL_K
70:        )
71:        retrieved_docs = [doc for doc, _ in results]
72:        scores = [score for _, score in results]
73:        confidence = sum(scores) / len(scores) if scores else 0.0
74:        metadata["retrieval_scores"] = [round(s, 4) for s in scores]
75:        metadata["retrieval_confidence"] = round(confidence, 4)
76:    else:
77:        retrieved_docs = vectorstore.similarity_search(search_text, k=RETRIEVAL_K)
78:
79:    sources = list(set(doc.metadata.get("source", "Unknown") for doc in retrieved_docs))
80:    context_text = "\n\n---\n\n".join(doc.page_content for doc in retrieved_docs)
81:
82:    # 4. Model routing
83:    if use_routing:
84:        selected_model, route_reason = select_generation_model(
85:            confidence=confidence,
86:            generation_models=settings.services.generation_models,
87:            confidence_threshold=CONFIDENCE_THRESHOLD,
88:        )
89:        metadata["route_reason"] = route_reason
90:        metadata["selected_model"] = selected_model.model
91:        generation_config = selected_model.model_dump()
92:    else:

97:    # 5. Generate
98:    system_instruction = (
99:        "You are a helpful, professional assistant. Use the following retrieved context "
